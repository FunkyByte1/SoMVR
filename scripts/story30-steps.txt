### fight2.sh
    #!/bin/bash
    # fight2.sh TAG HX HZ [HY] [ROUNDS] - som-b30-story5: the Uruks round Lithariel cut down at once (what a person's blade does
    # in seconds; the drill's teleport per orc took 4 s each): Talion 80 cm beside her, `fight health 1 all 700 not HER` (the
    # test aid; `not` read since build 30), the game's X every 0.9 s (4 per round) while an Uruk stays within 7 m of her
    TAG=$1; HX=$2; HZ=$3; HY=${4:-1350}; R=$(cat /tmp/b30s5/R-$TAG); L=$R/somvr.log
    wd() { printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    near() { local l0; l0=$(wc -l < $L); wd "verbs orcs $HX $HZ 700" 1.2; tail -n +$l0 $L | grep -aE '^VERBS orcs t=.* [0-9]+/' | grep -av "orc ${HER:-none} " | wc -l; }
    wd "practice go $((HX + 80)) $((HY + 50)) $HZ" 1.5
    for r in $(seq 1 ${5:-6}); do n=$(near); echo "$(date +%T) fight2 round $r: $n Uruks within 7 m of her"; [ "$n" -eq 0 ] && break
      l1=$(wc -l < $L); wd "fight health 1 all 700${HER:+ not $HER}" 1.0
      echo "  the aid on $(tail -n +$l1 $L | grep -ac '^FIGHT health'); her line: $(tail -n +$l1 $L | grep -a "^FIGHT health orc=${HER:-none} " | head -1 | cut -c1-80)"
      if tail -n +$l1 $L | grep -aq "^FIGHT health orc=${HER:-none} "; then wd "practice go $((HX + 60)) $((HY + 50)) $HZ" 1.2; wd "fight health 9999" 0.8; echo "  HER HIT: restored"; fi
      for k in 1 2 3 4; do wd "verbs press 118 150" 0.9; done
    done

### chain30.sh
    #!/bin/bash
    # chain30.sh TAG - som-b30-story5 boot: "The Rescue" to its end (rescue7.sh: the brands, find, the captors, her grab,
    # carry8.sh with the ally-safe aid, the end's screens), then the next missions' markers by mission3.sh (Torvin's next
    # 'goldenT' first, then a new 'goldenL'), each MIN minutes; then quit
    TAG=$1; R=$(cat /tmp/b30s5/R-$TAG); . /tmp/b30s5/lib.sh; L=$R/somvr.log; echo $$ > /tmp/b30s5/pid-dir-$TAG
    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    touch /tmp/b30s5/hold-$TAG
    echo "CHAIN markers at the start:"; markers '(golden[A-Z]|powerStruggle|confrontation[A-Za-z]*|resistance_QuestGiver)' | head -8
    if [ "${SKIP_RESCUE:-0}" = 0 ]; then echo "CHAIN rescue $(date +%T)"; CARRY_MIN=${CARRY_MIN:-20} /tmp/b30s5/rescue7.sh $TAG; echo "CHAIN rescue rc $? $(date +%T)"; fi
    wd "story invisible off" 0.5; wd "fight guard off" 0.3
    echo "CHAIN markers after it:"; M=$(markers '(golden[A-Z]|powerStruggle|confrontation[A-Za-z]*)'); echo "$M" | head -8
    for K in ${NEXT:-goldenT goldenL}; do
      P=$(markers "$K" | head -1 | cut -d' ' -f1-3); [ -n "$P" ] || { echo "CHAIN no $K"; continue; }
      echo "CHAIN $K at $P $(date +%T)"; /tmp/b30s5/mission3.sh $TAG "$P" "n-$K" ${MIN:-18}; echo "CHAIN $K done $(date +%T)"
      markers '(golden[A-Z]|powerStruggle)' | head -5
    done
    SYNC_S=60 sync "status" >/dev/null
    echo "CHAIN END $(date +%T)"; rm -f /tmp/b30s5/hold-$TAG; echo quit >> $R/cmd

### t2.sh
    #!/bin/bash
    # t2.sh TAG MIN - som-b30-story5: a story mission run as far as the harness goes, after Torvin: each round Torvin's
    # 'ObjectiveGuttered_ALLMaps' (else the newest 'Objective*' marker the game shows) by a trip beside it; a video: pictures,
    # A, B; a pausing tutorial or a window: pictures, A (then B if still up); enemies on an 'ObjectiveGold*' marker within
    # 40 m: the kill drill (behind, the aid, X); new subtitles / lessons / markers printed; MIN minutes or the SUCCESS
    TAG=$1; MIN=${2:-25}; R=$(cat /tmp/b30s5/R-$TAG); . /tmp/b30s5/lib.sh; L=$R/somvr.log; echo $$ > /tmp/b30s5/pid-dir-$TAG
    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { local x; { echo "# t2 $1"; shift; for x in "$@"; do echo "$x"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; sleep 0.3; }
    dump() { xt dump "dump"; sleep 1.2; echo "DUMP $1 $(grep -o 'dump layers frame [0-9]*' $R/xr-layer.log 2>/dev/null | tail -1 | grep -o '[0-9]*$')" | tee -a /tmp/b30s5/dumps-$TAG.txt; }
    win() { local a b; a=$(grep -an 'window 0 -> 1' $L | tail -1 | cut -d: -f1); b=$(grep -an 'window 1 -> 0' $L | tail -1 | cut -d: -f1); [ "${a:-0}" -gt "${b:-0}" ]; }
    news() { tail -n +$1 $L | grep -aE "^STORY t=.*subtitle ch=[01] '|^HINTS tutorial #.*show|^USE t=|Mission|^OBJECTIVE t=.*marker [0-9]+ m=.*shown by the game|^RIDE t=.*(offers|mounted|seated on)|^WARN" | cut -c1-170 | awk '!s[substr($0,index($0," ")+20)]++' | head -${2:-8}; }
    T=$((SECONDS + MIN * 60)); n=0; LB=$(wc -l < $L)
    while [ $SECONDS -lt $T ]; do n=$((n + 1)); l0=$(wc -l < $L); s=$(state)
      if echo "$s" | grep -q 'video=1'; then echo "$(date +%T) #$n video"; dump t2-v$n; SYNC_S=60 sync "wait 6" "pics t2-v$n" "tap R a 0.3" "tap R b 0.3" "wait 2" >/dev/null
      elif echo "$s" | grep -q 'paused=1' || win; then echo "$(date +%T) #$n window/paused ($s)"; sleep 2; dump t2-w$n; printf 'shot st-t2-w%s\n' $n >> $R/cmd; sleep 2
        xt a1 "btn R a 1"; sleep 0.3; xt a0 "btn R a 0"; sleep 2.5
        if win || state | grep -q 'paused=1'; then printf 'shot st-t2-w%s-b\n' $n >> $R/cmd; sleep 1.5; xt b1 "btn R b 1"; sleep 0.3; xt b0 "btn R b 0"; sleep 2.5; fi
      else
        wd "objective status" 1.2; wd "practice status" 0.6
        TV=$(tail -n +$l0 $L | grep -aE "^OBJECTIVE status marker .*'ObjectiveGuttered_ALLMaps'" | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
        OB=$(tail -n +$l0 $L | grep -aE "^OBJECTIVE status marker .*'(Objective|ObjectiveGold[A-Za-z]*)' " | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
        read px py pz < <(tail -n +$l0 $L | grep -aoE 'Talion known at \(-?[0-9.]+ -?[0-9.]+ -?[0-9.]+\)' | tail -1 | grep -oE -- '-?[0-9.]+' | awk 'NR==1{x=$1} NR==2{y=$1} NR==3{printf "%d %d %d\n",x,y,$1}')
        G=${TV:-$OB}; [ $((n % 6)) = 1 ] && echo "$(date +%T) #$n Talion ${px:-?} ${pz:-?} | Torvin ${TV:-none} | objective ${OB:-none}"
        if [ -n "$G" ]; then read gx gy gz <<< "$G"; d=$(awk -v a=$gx -v b=$gz -v x=${px:-0} -v z=${pz:-0} 'BEGIN{print int(sqrt((a-x)^2+(b-z)^2))}')
          if [ "$d" -gt 400 ]; then SYNC_S=60 sync "go $((gx + 150)) $((gy + 100)) $gz" "wait 2" >/dev/null; else sleep 3; fi
        else sleep 3; fi
        [ $((n % 15)) = 0 ] && dump t2-n$n
      fi
      news $l0 6
      tail -n +$LB $L | grep -aqE "SUCCESS|Mission Complete" && { echo "$(date +%T) success?"; }
      [ -n "${STOP_RE:-}" ] && tail -n +$LB $L | grep -aqE "$STOP_RE" && { echo "$(date +%T) T2 STOP: $(tail -n +$LB $L | grep -aE "$STOP_RE" | head -1 | cut -c1-120)"; break; }
    done
    echo "T2 END $(date +%T) $(state)"; SYNC_S=60 sync "status" "word practice status nav" >/dev/null
    grep -aE "^PRACTICE nav [0-9]+ '(golden|Objective|power|quest|Quest|mission|resistance)" $L | tail -6 | cut -c1-120
