### boot3.sh
    #!/bin/bash
    # boot2.sh N - like boot.sh; the director in $DIRECTOR (not in the command line), the save copy in $SAVE_FROM
    N=$1; WT=/home/bara/.t3/worktrees/flat2vr/t3-4e8f4da5/ShadowOfMordor
    while :; do if [ -e /tmp/somvr-boot.lock ]; then sleep 2; continue; fi
      /tmp/b27s3/gate.sh; rc=$?; [ $rc = 0 ] && break; [ $rc = 4 ] && { echo "amdgpu reset: stop"; exit 4; }; sleep 2; done
    cd $WT
    ( sleep 5; /tmp/b27s3/run.sh q$N "$DIRECTOR" > /tmp/b27s3/dir-q$N.log 2>&1 ) &
    HEADLESS_SAVE_FROM=${SAVE_FROM:-} HEADLESS_HOLD=${HOLD:-1700} SOMVR_EXTRA="$(grep -E '^SOMVR_[A-Z0-9_]+=' scripts/headset-build26.env | tr '\n' ' ') SOMVR_PRACTICE=1 SOMVR_PRACTICE_QUIET=0 SOMVR_PRACTICE_CLEAR=0 SOMVR_PRACTICE_DAY=0 SOMVR_HANDS_TEST=1 SOMVR_XR_TEST=@RUN@/xrtest.txt SOMVR_EYE_SIZE=2015x2217 ${BOOT_EXTRA:-}" setsid -w scripts/headless_vr.sh $WT/mod/somvr > /tmp/b27s3/boot$N.log 2>&1
    echo "boot exit $?"; grep -q 'som-b27-story3' /tmp/somvr-boot.lock/owner && rm -rf /tmp/somvr-boot.lock; echo "released $(date +%T)" | tee -a /tmp/b27s3/release.log
    wait

### run.sh
    #!/bin/bash
    # run.sh TAG DIRECTOR [ARGS] - after the boot: the driver (story22-drive.sh ctl, queue /tmp/b27s3/ctl-TAG) + the director
    TAG=$1; DIR=$2; shift 2; WT=/home/bara/.t3/worktrees/flat2vr/t3-4e8f4da5/ShadowOfMordor
    rm -rf /tmp/b27s3/ctl-$TAG; mkdir -p /tmp/b27s3/ctl-$TAG
    cd $WT/scripts && ST_PLAN=ctl ST_CTL_DIR=/tmp/b27s3/ctl-$TAG ./story22-drive.sh > /tmp/b27s3/drv-$TAG.log 2>&1 &
    for i in $(seq 1 200); do grep -q '^run ' /tmp/b27s3/drv-$TAG.log && break; sleep 3; done
    R=$(grep '^run ' /tmp/b27s3/drv-$TAG.log | head -1 | cut -c5-); [ -n "$R" ] || { echo "no run"; exit 1; }
    echo "$R" > /tmp/b27s3/R-$TAG; echo "R=$R"
    for i in $(seq 1 300); do grep -q 'waiting for' /tmp/b27s3/drv-$TAG.log && break; grep -q 'STOP' /tmp/b27s3/drv-$TAG.log && exit 1; sleep 2; done
    ( while sleep 240; do [ -d /proc/$$ ] || exit 0; /tmp/b27s3/q.sh $TAG "wait 1"; done ) & KA=$!  # the driver quits after 15 idle minutes
    "$DIR" $TAG "$@"; kill $KA 2>/dev/null

### gate.sh
    #!/bin/bash
    # gate.sh - the VR gate + the boot lock for som-b27-story3; exit 0 = the lock is ours
    [ "$(systemctl --user is-active somvr-headset 2>/dev/null)" = active ] && { echo "gate: somvr-headset active (user in VR)"; exit 2; }
    grep -q '"user": false' /run/user/$(id -u)/vr-session/status.json 2>/dev/null || { echo "gate: status.json user not false"; exit 2; }
    pgrep -x monado-service >/dev/null && { echo "gate: monado-service running"; exit 1; }
    pgrep -f '[H]and of Fate 2\.' >/dev/null && { echo "gate: HoF2 running"; exit 1; }
    journalctl -k -b --no-pager 2>/dev/null | grep -qiE 'amdgpu.*(reset|VRAM is lost)' && { echo "gate: amdgpu reset in the kernel log"; exit 4; }
    mkdir /tmp/somvr-boot.lock 2>/dev/null || { echo "gate: lock busy ($(cat /tmp/somvr-boot.lock/owner 2>/dev/null))"; exit 1; }
    echo som-b27-story3 > /tmp/somvr-boot.lock/owner; echo "gate: ok $(date +%T)"; exit 0

    # brand9.sh TAG N - "The Rescue"'s camp: brand7.sh's spots, then rounds over the camp's live orcs not branded by us yet
    # (pointers), until N brands (the game's 'HierarchyWhiteHand' markers since the start) or 3 rounds

      DONE=$(grep -oE 'orc 0x[0-9a-f]+ at .*brand 1' /tmp/b27s3/brand9-$TAG.log | grep -oE '0x[0-9a-f]+' | tr '\n' '|'); DONE=${DONE%|}

      echo "$(date +%T) round $round: brands $(nb), candidates $S"; [ -n "$S" ] || break
      eval "set -- $S"; SP=""; for s in "$@"; do SP="$SP \"$s\""; done
      sed "s#^for o in \"-27223 24205\".*; do#for o in $SP; do#" /tmp/b27s3/brand7.sh > /tmp/b27s3/brand9r.sh; chmod +x /tmp/b27s3/brand9r.sh

    done
    echo "$(date +%T) brands $(nb)"; [ "$(nb)" -ge $N ]

### brand7.sh
    #!/bin/bash
    # brand7.sh TAG N - "The Rescue"'s camp: N sentries branded by the left hand + the trigger at the game's offer; per orc
    # (the nearest to each spot, not one branded since the start = a new 'HierarchyWhiteHand' marker within 2.5 m): from the
    # front at 150 / 110 cm, then from behind at 120 cm (the shield sentry), each the trigger again until the brand starts
    TAG=$1; N=${2:-6}; R=$(cat /tmp/b27s3/R-$TAG); L=$R/somvr.log
    wd() { printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { { echo "# b7 $1"; shift; for l in "$@"; do echo "$l"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; }
    M0=$(wc -l < $L)
    marks() { tail -n +$M0 $L | grep -aE "marker [0-9]+ m=.*'HierarchyWhiteHand': shown" | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4- | awk '{print $1, $3}'; }
    for o in "-27223 24205" "-26754 24245" "-27361 24509" "-26935 24625" "-27502 24549" "-27443 26369" "-27633 27215" "-27804 30831" "-25229 24510"; do
      set -- $o; l0=$(wc -l < $L); wd "verbs orcs $1 $2 700" 1.3
      LN=$(tail -n +$l0 $L | grep -E '^VERBS orcs t=.* [0-9]+/' | head -1); read oc ox oy oz f < <(echo "$LN" | sed -E 's/.* orc (0x[0-9a-f]+) at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing (-?[0-9.]+) deg.*/\1 \2 \3 \4 \5/')
      [ -n "$f" ] || { echo "no orc at $o"; continue; }
      marks | awk -v X=$ox -v Z=$oz '(($1-X)^2+($2-Z)^2) < 250^2 {f=1} END {exit !f}' && { echo "orc at $ox $oz: branded already"; continue; }
      ok=0
      for D in 150 110 -120; do
        read tx tz < <(python3 -c "import math;f=math.radians($f);print(round($ox+$D*math.sin(f)),round($oz+$D*math.cos(f)))")
        wd "crouch hold 10000"; wd "verbs aim head"; b0=$(stat -c %s $L); wd "practice go $tx $(( oy + 60 )) $tz" 0.1
        for i in $(seq 1 60); do tail -c +$b0 $L | grep -qE "prompt up: action 120 .*Stealth_Brand" && break; sleep 0.05; done
        for k in 1 2 3; do sleep 0.3; xt 1 "btn L trigger 1"; sleep 0.25; xt 2 "btn L trigger 0"; sleep 1.0
          tail -c +$b0 $L | grep -qE "Talion attack 'PC_(Bip|Archer)_StealthBrand" && { ok=1; break; }; done
        [ $ok = 1 ] && break
        tail -c +$b0 $L | grep -qE "window 0 -> 1" && { echo "a window: stop"; exit 2; }
      done
      sleep 2.5; n=$(marks | wc -l); echo "$(date +%T) orc $oc at $ox $oz: brand $ok (D $D), brands since the start $n"
      [ $n -ge $N ] && break
    done
    printf 'shot st-rs-brands7\n' >> $R/cmd

### rescue3.sh
    #!/bin/bash
    # rescue3.sh TAG - som-b27-story3: "The Rescue" (Lithariel) start to end from 070839's save. The marker (B), the card (A),
    # the video; the camp's 6 sentries branded by the hand (brand6.sh); "Find Lithariel"; her 5 captors: Talion 100 cm

    # as boot 055352), the walk until the failed screen, A with the grip still closed = Retry; then the real carry (her full

    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { local x; { echo "# r3 $1"; shift; for x in "$@"; do echo "$x"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; sleep 0.3; }
    shot() { printf 'shot st-%s\n' "$1" >> $R/cmd; sleep 1.3; }

    ev() { tail -n +$1 $R/somvr.log | grep -E "^STORY t=.*subtitle ch=[01] '|^HINTS tutorial #.*show|^USE t=|window 0 -> 1|Talion attack 'PC_Grab[A-Za-z_]*|^FIGHT health|^OBJECTIVE t=.*marker [0-9]+ m=.*shown|^WARN" | cut -c1-170 | awk '!s[$0]++' | head -${2:-10}; }
    nav() { local l0; l0=$(wc -l < $R/somvr.log); wd "practice status nav" 1.5; tail -n +$l0 $R/somvr.log | grep -E "^PRACTICE nav [0-9]+ '$1'" | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\), [0-9.]+ m' | sed -E 's/at \(//; s/\),//; s/ m$//' | sort -k4 -n; }
    LX=${LITH_X:--28266}; LZ=${LITH_Z:-33695}   # where she kneels (boot 055352: orc 0x40e645c8, all 16 lists)
    L0=$(wc -l < $R/somvr.log); wd "story invisible on" 0.5; wd "hands test R 25 -45 -25 0 0 0" 0.5
    if [ "${SKIP_START:-0}" = 0 ]; then
    echo "PHASE start $(date +%T)"

    echo "PHASE brands $(date +%T)"

    sleep 3; shot rs-brands; L=$(wc -l < $R/somvr.log)
    O=$(nav Objective | head -1); echo "$(date +%T) Find Lithariel: ${O:-none}"; set -- ${O:--28058 1850 33551}
    SYNC_S=180 sync "go $1 1400 $3" "waitfor SESSION.ui.*video=1 30" "wait 3" "tap R a 0.3" "tap R b 0.3" "world 60" "wait 2" "pics rs-captors" >/dev/null
    ev $L 8
    fi
    echo "PHASE captors $(date +%T)"
    L=$(wc -l < $R/somvr.log)
    for pass in $( [ "${SKIP_CAPTORS:-0}" = 1 ] || echo 1 2 3 4 5 6); do
      l0=$(wc -l < $R/somvr.log); wd "verbs orcs $LX $LZ 3000" 1.2

        xt g0 "btn L squeeze 0"; sleep 1.5; done; wd "verbs aim off" 0.3; return 1; }

    [ "${NO_CARRY:-0}" = 1 ] && { echo "$(date +%T) held: NO_CARRY"; exit 0; }
    if [ "${GRAB_TEST:-1}" = 1 ]; then
      echo "PHASE grabtest $(date +%T)"; L=$(wc -l < $R/somvr.log)
      wd "fight health 1" 0.6; tail -n +$L $R/somvr.log | grep -E '^FIGHT health' | head -1 | cut -c1-120
      S=$(nav Objective | head -1); set -- ${S:--29190 1554 30650}
      python3 $WT/scripts/story2_route.py "$R" --secs 70 --tag rsg --nosprint --noedge --learn 0.3 --waypoints "$1,$3;-27404,23277" 2>&1 | tail -3 | cut -c1-170

      echo "after A (the grip still closed):"; tail -n +$L2 $R/somvr.log | grep -E "^USE t=|QuestFailedRetry|^VERBS t=.*(release|no stereo)|window" | head -6 | cut -c1-170
      xt g2 "btn L squeeze 0"; sleep 1
      SYNC_S=120 sync "world 90" "wait 3" "pics rs-retry" >/dev/null; echo "$(date +%T) after the retry: $(state)"
      L=$(wc -l < $R/somvr.log); grab || { echo "no grab after the retry"; exit 2; }
    fi
    echo "PHASE carry $(date +%T)"; L=$(wc -l < $R/somvr.log)
    for leg in 1 2 3 4 5; do
      S=$(grep -E "^OBJECTIVE t=.*marker [0-9]+ m=.* 'Objective'.* shown by the game at" $R/somvr.log | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
      [ -n "$S" ] || S=$(nav Objective | head -1 | cut -d' ' -f1-3); set -- $S; echo "$(date +%T) leg $leg toward $S"
      LG=$(wc -l < $R/somvr.log)
      python3 $WT/scripts/story2_route.py "$R" --secs ${LEG_S:-90} --tag rsc$leg --nosprint --noedge --learn 0.3 --dumps 20 --waypoints "$1,$3" 2>&1 | tail -3 | cut -c1-170
      ev $L 14; L=$(wc -l < $R/somvr.log); dump rs-leg$leg
      if grep -E 'window 0 -> 1' $R/somvr.log | tail -1 | grep -q .; then a=$(grep -n 'window 0 -> 1' $R/somvr.log | tail -1 | cut -d: -f1); b=$(grep -n 'window 1 -> 0' $R/somvr.log | tail -1 | cut -d: -f1); [ "${a:-0}" -gt "${b:-0}" ] && { echo "a window: stop the legs"; break; }; fi
      tail -n +$LG $R/somvr.log | grep -qE "VERBS t=.*release after|no stereo frame read the left hand" && { echo "the grab let go: stop the legs"; break; }
    done
    echo "PHASE end $(date +%T)"
    xt g3 "btn L squeeze 0"; sleep 1
    /tmp/b27s3/endphase.sh $TAG 6 rs-end
    echo "$(date +%T) rescue3 done: $(state)"

      mapfile -t O < <(tail -n +$l0 $L | grep -aE '^VERBS orcs t=.* [0-9]+/' | sed -E 's/.* orc (0x[0-9a-f]+) at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing.*/\1 \2 \3 \4/' | awk -v X=$HX -v Z=$HZ -v H=${HER:-none} '$1 != H && (H != "none" || (($2-X)^2+($4-Z)^2) > 250^2)')
      echo "$(date +%T) pass $pass: ${#O[@]} Uruks"; [ ${#O[@]} -eq 0 ] && break
      for o in "${O[@]}"; do read oc ox oy oz <<< "$o"
        read tx tz < <(python3 -c "import math;dx=$ox-($HX);dz=$oz-($HZ);d=math.hypot(dx,dz) or 1;print(round($ox+100*dx/d),round($oz+100*dz/d))")

        wd "verbs press 118 150" 1.3; wd "verbs press 118 150" 1.2
      done
    done

    #!/bin/bash
    # endphase.sh TAG MIN NAME - after the last objective: videos (2 pictures, then A, B), paused menus (a picture, A), the
    # window (pictures, then A, then B), until the world is quiet for 6 rounds or MIN minutes
    TAG=$1; MIN=${2:-6}; NM=${3:-end}; R=$(cat /tmp/b27s3/R-$TAG); . /tmp/b27s3/lib.sh
    win() { local a b; a=$(grep -n 'window 0 -> 1' $R/somvr.log | tail -1 | cut -d: -f1); b=$(grep -n 'window 1 -> 0' $R/somvr.log | tail -1 | cut -d: -f1); [ "${a:-0}" -gt "${b:-0}" ]; }
    q=0; n=0; T=$((SECONDS + MIN * 60))
    while [ $SECONDS -lt $T ]; do n=$((n + 1)); s=$(state)
      if echo "$s" | grep -q 'video=1'; then q=0; echo "$(date +%T) $n video"; SYNC_S=90 sync "pics $NM-v$n" "wait 8" "pics $NM-v$n-b" "tap R a 0.3" "tap R b 0.3" "wait 2"
      elif echo "$s" | grep -q 'paused=1'; then q=0; echo "$(date +%T) $n paused"; SYNC_S=60 sync "wait 2" "pics $NM-p$n" "tap R a 0.4" "wait 2"
      elif win; then q=0; echo "$(date +%T) $n window"; SYNC_S=60 sync "pics $NM-w$n" "wait 6" "pics $NM-w$n-b"; win && SYNC_S=60 sync "tap R a 0.4" "wait 3"; qf=$(grep -an "questfail_in\|command 285" $R/somvr.log | tail -1 | cut -d: -f1); wo=$(grep -an "window 0 -> 1" $R/somvr.log | tail -1 | cut -d: -f1); [ "${qf:-0}" -gt "$(( ${wo:-0} - 40 ))" ] && echo "a failed screen: A only" || { win && SYNC_S=60 sync "tap R b 0.4" "wait 3"; }
      else q=$((q + 1)); [ $q -ge 6 ] && { echo "$(date +%T) quiet: done"; break; }; SYNC_S=60 sync "wait 4" "pics $NM-q$n" >/dev/null; fi
      grep -E "^STORY t=.*subtitle ch=[01] '|^HINTS tutorial #.*show|^USE t=" $R/somvr.log | tail -2 | cut -c1-150
    done
    SYNC_S=60 sync "status" "word practice status nav" >/dev/null
    grep -E "^PRACTICE nav [0-9]+ '(golden|Objective|power|quest|Quest|mission)" $R/somvr.log | tail -12 | cut -c1-120

    #!/bin/bash
    # next3.sh TAG [WHAT...] - som-b27-story3: after the story missions, what the save offers: the markers (a survey), the map
    # challenges (herbs: each kind of flower near picked by B at its prompt; the map's challenge panel by the right stick's
    # click), then the Nemesis missions from their markers (mission3.sh: the marker's B, the card, the screens, the harness's
    # kill drill). WHAT: herbs, map, M:<marker name> (default: survey herbs map M:confrontationBranded M:powerStruggle)
    TAG=$1; shift; R=$(cat /tmp/b27s3/R-$TAG); . /tmp/b27s3/lib.sh; echo $$ > /tmp/b27s3/pid-dir-$TAG; L=$R/somvr.log
    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { local x; { echo "# nx $1"; shift; for x in "$@"; do echo "$x"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; sleep 0.3; }
    shot() { printf 'shot st-%s\n' "$1" >> $R/cmd; sleep 1.3; }

    paused() { grep -E '^SESSION ui ' $L | tail -1 | grep -q 'paused=1'; }
    navall() { local l0; l0=$(wc -l < $L); wd "practice status nav" 1.5; tail -n +$l0 $L | grep -E "^PRACTICE nav [0-9]+ '"; }
    xyz() { grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\), [0-9.]+ m' | sed -E 's/at \(//; s/\),//; s/ m$//' | sort -k4 -n; }
    out() { for k in 1 2 3 4; do paused || break; xt o1 "btn R b 1"; sleep 0.2; xt o2 "btn R b 0"; sleep 2; done; }
    [ $# -eq 0 ] && set -- survey herbs map M:confrontationBranded M:powerStruggle
    wd "story invisible on" 0.5; wd "hands test R 25 -45 -25 0 0 0" 0.5
    for what in "$@"; do echo "PHASE $what $(date +%T)"
      case $what in
      survey) N=$(navall); echo "$N" | grep -vE "Collectible|RessurrectionStone|FT_" | grep -oE "'[A-Za-z_0-9]+' at \([-0-9 ]+\), [0-9.]+ m" | awk '!s[$1]++' | head -20
        echo "$N" | grep -oE "'Collectible_Flower_[A-Z]'" | sort | uniq -c | tr '\n' ' '; echo ;;
      herbs) N=$(navall)

          l0=$(wc -l < $L); SYNC_S=60 sync "go $1 $(( $2 + 60 )) $3" "wait 2" >/dev/null
          for i in $(seq 1 20); do tail -n +$l0 $L | grep -q "USE t=.*pick prompt up" && break; sleep 0.25; done
          P2=$(tail -n +$l0 $L | grep -m1 "USE t=.*pick prompt up" | cut -c1-140); shot herb-$f-at
          xt b1 "btn R b 1"; sleep 1.2; xt b0 "btn R b 0"; sleep 2.5; shot herb-$f
          echo "$f at $1 $3 ($4 m): ${P2:-no pick prompt}"; tail -n +$l0 $L | grep -E "USE t=.*(B at|pick prompt down)|HealthCollectable|Collectable|Niphredil|Challenge" | head -3 | cut -c1-150
        done ;;
      map) l0=$(wc -l < $L); xt r1 "btn R stick 1"; sleep 0.2; xt r2 "btn R stick 0"; sleep 3; shot map-1; dump map-1
        tail -n +$l0 $L | grep -E "^INPUT t=.*map |SESSION ui" | head -2 | cut -c1-150; sleep 1; out ;;
      M:*) nm=${what#M:}; P=$(navall | grep "'$nm'" | xyz | head -1); [ -n "$P" ] || { echo "no '$nm' marker"; continue; }
        set -- $P; echo "$nm at $1 $2 $3 ($4 m)"; KILL=${KILL:-1} /tmp/b27s3/mission3.sh $TAG "$1 $2 $3" ${nm:0:6} ${MIN_M:-12}
        out; dump $nm-after ;;
      esac
    done
    echo "$(date +%T) next3 done: $(state)"
