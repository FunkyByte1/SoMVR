    #!/bin/bash
    # nm9.sh TAG - from the furthest save (106.7): the markers; "No Warchief Selected - Hold [>]": the game's held commands
    # one by one (14 Menu = START 1.5 s, then 14 tapped 0.2 s, 281 EnterHierarchyMenu, 95 PDA = BACK, 316 InformationShortcut; then
    # 1.5 s each), a picture and the session state after each; then Marwen's next mission from its marker (mission3.sh)
    TAG=$1; R=$(cat /tmp/b27s2/R-$TAG); . /tmp/b27s2/lib.sh; echo $$ > /tmp/b27s2/pid-dir-$TAG
    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { local x; { echo "# n $1"; shift; for x in "$@"; do echo "$x"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; sleep 0.3; }
    shot() { printf 'shot st-%s\n' "$1" >> $R/cmd; sleep 1.3; }
    paused() { grep -E '^SESSION ui ' $R/somvr.log | tail -1 | grep -q 'paused=1'; }
    ev() { tail -n +$1 $R/somvr.log | grep -E "^STORY t=.*subtitle ch=[01] '|^HINTS tutorial #.*show|^USE t=|window 0 -> 1|SESSION ui|^TEXTBUTTONS|^VRMENU|shown by the game|^WARN|EnterHierarchy|Hierarchy|^VERBS t=.*press" | cut -c1-170 | awk '!s[$0]++' | head -${2:-10}; }
    wd "story invisible on" 0.5; wd "hands test R 25 -45 -25 0 0 0" 0.5
    L=$(wc -l < $R/somvr.log); wd "practice status nav" 1.5
    tail -n +$L $R/somvr.log | grep -E "^PRACTICE nav [0-9]+ '" | grep -vE "Collectible|Flower|Ressur|FT_" | grep -oE "'[A-Za-z_]+' at \([-0-9 ]+\), [0-9.]+ m" | awk '!s[$1]++' | head -16
    M=$(tail -n +$L $R/somvr.log | grep -E "^PRACTICE nav [0-9]+ 'goldenM'" | head -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
    sleep 2; shot nm-arrival; echo "$(date +%T) arrival: $(state)"
    for c in 14 14h5 14h9 14t 281 95 316; do L=$(wc -l < $R/somvr.log)
      case $c in 14t) wd "verbs press 14 200" 2.0;; 14h5) wd "verbs press 14 500" 2.0;; 14h9) wd "verbs press 14 900" 2.0;; *) wd "verbs press $c 1500" 2.6;; esac
      sleep 1; shot nm-hold-$c; echo "$(date +%T) the hold by $c: $(state)"; ev $L 8
      sleep 2; shot nm-hold-$c-b
      paused && { xt b1 "btn R b 1"; xt b2 "btn R b 0"; sleep 2.5; }; paused && { xt b3 "btn R b 1"; xt b4 "btn R b 0"; sleep 2.5; }
      paused && { xt b5 "btn L menu 1"; sleep 0.2; xt b6 "btn L menu 0"; sleep 2.5; }; echo "   after: $(state)"
    done
    # SOMVR_ARMY (BOOT_EXTRA): the right stick's click 0.2 s (the map at the release), then held 1.0 s (the game's Hold [>])
    for k in click hold; do L=$(wc -l < $R/somvr.log); xt r1 "btn R stick 1"; [ $k = click ] && sleep 0.2 || sleep 1.0; xt r2 "btn R stick 0"; sleep 2.5
      shot nm-r3-$k; echo "$(date +%T) the right stick $k: $(state)"; ev $L 8; tail -n +$L $R/somvr.log | grep -E "^INPUT t=.*(army|map) " | head -2 | cut -c1-170
      sleep 2; shot nm-r3-$k-b
      paused && { xt b1 "btn R b 1"; xt b2 "btn R b 0"; sleep 2.5; }; paused && { xt b3 "btn R b 1"; xt b4 "btn R b 0"; sleep 2.5; }
      paused && { xt b5 "btn L menu 1"; sleep 0.2; xt b6 "btn L menu 0"; sleep 2.5; }; echo "   after: $(state)"
    done
    echo "$(date +%T) Marwen's next marker: ${M:-none}"; [ -n "$M" ] || exit 0
    /tmp/b27s2/mission3.sh $TAG "$M" nm 17

### area.sh
    #!/bin/bash
    # area.sh TAG RR X Z [X Z...] - each area: the orcs within RR cm of X,Z (`verbs orcs`), each one: Talion 100 cm behind
    # it, its health 1, the game's own X (118) twice (the stealth kill / a strike); then a picture
    TAG=$1; RR=$2; shift 2; R=$(cat /tmp/b27s2/R-$TAG); . /tmp/b27s2/lib.sh
    wd() { userin && { echo "USER IN VR: stop"; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    wd "hands test R 25 -45 -25 0 0 0"
    while [ $# -ge 2 ]; do X=$1; Z=$2; shift 2
      for pass in 1 2; do
        a0=$(grep -c 'arrived at' $R/somvr.log); wd "practice go $X ${AREA_Y:-4000} $Z" 0.3
        for i in $(seq 1 40); do [ "$(grep -c 'arrived at' $R/somvr.log)" -gt "$a0" ] && break; sleep 0.25; done
        l0=$(wc -l < $R/somvr.log); wd "verbs orcs $X $Z $RR" 1.2
        mapfile -t O < <(tail -n +$l0 $R/somvr.log | grep -E '^VERBS orcs t=.* [0-9]+/' | grep -v "dead\|health 0" | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing.*/\1 \3/')
        echo "$(date +%T) area $X $Z pass $pass: ${#O[@]} orcs"
        [ ${#O[@]} -eq 0 ] && break
        for o in "${O[@]}"; do set -- $o $@; ox=$1; oz=$2; shift 2
          l1=$(wc -l < $R/somvr.log); wd "verbs behind 100 $ox $oz" 1.0; [ "${AREA_HP:-1}" = 1 ] && wd "fight health 1"; wd "verbs press 118 150" 1.4; wd "verbs press 118 150" 1.2
          tail -n +$l1 $R/somvr.log | grep -E "^FIGHT health|attack 'PC_[A-Za-z_]*(Kill|Strike|Instakill)" | head -2 | cut -c1-120
        done
      done
      printf 'shot st-area-%s\n' "$X" >> $R/cmd; sleep 1.5
      grep -E "^STORY t=.*subtitle ch=1 '|Mission|alarm|Alarm" $R/somvr.log | tail -2 | cut -c1-150
    done

### brand6.sh
    #!/bin/bash
    # brand6.sh TAG N - "The Rescue": the camp's sentries branded by the hand (front.sh's approach, the trigger pressed again
    # until the game's brand starts, 3 tries), until N brands since the start
    TAG=$1; N=${2:-6}; R=$(cat /tmp/b27s2/R-$TAG)

    xt() { { echo "# b6 $1"; shift; for l in "$@"; do echo "$l"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; }
    c0=$(grep -cE "sync 'PC_(Bip|Archer)_StealthBrand" $R/somvr.log); done_n=0
    for o in "-27214 24202" "-26748 24240" "-27523 24536" "-27363 24537" "-26774 24549" "-26923 24639" "-27438 26371" "-27310 27953" "-27899 30798"; do
      set -- $o; l0=$(wc -l < $R/somvr.log); wd "verbs orcs $1 $2 900" 1.5
      L=$(tail -n +$l0 $R/somvr.log | grep -E '^VERBS orcs t=.* [0-9]+/' | head -1); read ox oy oz f < <(echo "$L" | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing (-?[0-9.]+) deg.*/\1 \2 \3 \4/')
      [ -n "$f" ] || { echo "no orc at $o"; continue; }
      for D in 150 110; do
        read tx tz < <(python3 -c "import math;f=math.radians($f);print(round($ox+$D*math.sin(f)),round($oz+$D*math.cos(f)))")
        wd "crouch hold 10000"; wd "verbs aim head"; b0=$(stat -c %s $R/somvr.log); wd "practice go $tx $(( oy + 60 )) $tz" 0.1
        for i in $(seq 1 100); do tail -c +$b0 $R/somvr.log | grep -qE "prompt up: action 120 .*Stealth_Brand" && break; sleep 0.05; done
        ok=0; for k in 1 2 3; do sleep 0.3; xt 1 "btn L trigger 1"; sleep 0.25; xt 2 "btn L trigger 0"; sleep 1.0
          tail -c +$b0 $R/somvr.log | grep -qE "Talion attack 'PC_(Bip|Archer)_StealthBrand" && { ok=1; break; }; done

      done
      n=$(( $(grep -cE "sync 'PC_(Bip|Archer)_StealthBrand" $R/somvr.log) - c0 )); echo "$(date +%T) orc at $ox $oz: brand ${ok} (D $D), the brand syncs since the start $n"; sleep 2.5

    TAG=$1; X=$2; Z=$3; D=${4:-120}; R=$(cat /tmp/b27s2/R-$TAG)
    wd() { printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { { echo "# bh $1"; shift; for l in "$@"; do echo "$l"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; }
    wd "crouch hold 10000"; wd "verbs aim head"; b0=$(stat -c %s $R/somvr.log); wd "verbs behind $D $X $Z" 0.1
    for i in $(seq 1 160); do tail -c +$b0 $R/somvr.log | grep -qE "prompt up: action 120 .*Stealth_Brand" && break; sleep 0.05; done
    xt 1 "btn L trigger 1"; sleep 0.2; xt 2 "btn L trigger 0"; echo "behind $X $Z: prompt after $i polls"; sleep 3
    tail -c +$b0 $R/somvr.log | grep -E "attack 'PC_(Archer|Bip)_StealthBrand|no brand offered|offers the stealth" | head -2 | cut -c1-150
