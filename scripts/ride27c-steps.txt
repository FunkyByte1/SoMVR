### trace.awk
    # trace.awk T0 T1: the RIDE view lines between t T0 and T1, every 4th frame: t, the beast's speed (m/s over 4 frames), y, dy
    /^RIDE view f=/ { split($4,a,"="); t=a[2]+0; if (t < T0+0 || t > T1+0) next
      split($7,p,","); x=p[1]+0; y=p[2]+0; z=p[3]+0; k++
      if (k % 4 == 1) { if (k > 1) printf "%.2f v=%.1f y=%.0f dy=%+.0f\n", t, sqrt((x-px)^2+(z-pz)^2)/(t-pt)/100, y, y-py; pt=t; px=x; py=y; pz=z } }

### speeds.awk
    # speeds.awk - the beast's (Talion's root) horizontal speed between ANCHOR lines (frame numbers at ~20 Hz), by which
    # of 16 / 184 / 18 is held (VIN press/release) and whether the stick is on (cmd 3/6)
    /^VIN press cmd=(16|184|18|3|6) / { split($3,a,"="); h[a[2]]=1 }
    /^VIN release cmd=(16|184|18|3|6) / { split($3,a,"="); h[a[2]]=0 }
    /^RATE t=/ { split($2,a,"="); t=a[2] }
    /^ANCHOR frame=/ { split($2,a,"="); f=a[2]; split($4,p,","); x=p[1]; z=p[3]
      if (pf && f>pf && f-pf<=25) { d=sqrt((x-px)^2+(z-pz)^2); v=d/((f-pf)/20)/100
        k=(h[3]||h[6]?"stick":"nostick") (h[16]?"+A16":"") (h[184]?"+C184":"") (h[18]?"+F18":"")
        if (v<40) { n[k]++; s[k]+=v; if(v>m[k])m[k]=v; if (v>=12) fast[k]++ } }
      pf=f; px=x; pz=z }
    END { for (k in n) printf "%-28s n=%4d mean %5.1f m/s max %5.1f  >=12m/s %d\n", k, n[k], s[k]/n[k], m[k], fast[k] }

        OB=$(tail -n +$l0 $L | grep -aE "^OBJECTIVE status marker .*'Objective' " | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
        read px pz < <(tail -n +$l0 $L | grep -aoE 'Talion known at \(-?[0-9.]+ -?[0-9.]+ -?[0-9.]+\)' | tail -1 | grep -oE -- '-?[0-9.]+' | awk 'NR==1{x=$1} NR==3{printf "%d %d\n",x,$1}')
