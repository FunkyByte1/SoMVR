### gate2.sh
    #!/bin/bash
    # gate2.sh - wait for the boot lock (mkdir every 0.05 s), then the VR gate; exit 0 = the lock is ours and the gate open
    while :; do
      until mkdir /tmp/somvr-boot.lock 2>/dev/null; do sleep 0.05; done
      echo som-b27-story4 > /tmp/somvr-boot.lock/owner
      ok=1
      [ "$(systemctl --user is-active somvr-headset 2>/dev/null)" = active ] && { echo "gate: user in VR"; ok=0; }
      grep -q '"user": false' /run/user/$(id -u)/vr-session/status.json 2>/dev/null || { echo "gate: status.json user not false"; ok=0; }
      for i in $(seq 1 40); do pgrep -x monado-service >/dev/null || break; sleep 0.5; done   # the last boot's monado ending
      pgrep -x monado-service >/dev/null && { echo "gate: monado-service running"; ok=0; }
      pgrep -f '[H]and of Fate 2\.' >/dev/null && { echo "gate: HoF2 running"; ok=0; }
      journalctl -k -b --no-pager 2>/dev/null | grep -qiE 'amdgpu.*(reset|VRAM is lost)' && { echo "gate: amdgpu reset"; grep -q som-b27-story4 /tmp/somvr-boot.lock/owner && rm -rf /tmp/somvr-boot.lock; exit 4; }
      [ $ok = 1 ] && { echo "gate: ok $(date +%T)"; exit 0; }
      grep -q som-b27-story4 /tmp/somvr-boot.lock/owner && rm -rf /tmp/somvr-boot.lock; sleep 60
    done

### remount.sh
    #!/bin/bash
    # remount.sh TAG - on foot near the mission's caragor (not Torvin's beckoning one): go beside it, face it, the left hand
    # out, the offer, the grip closed; 3 tries
    TAG=$1; R=$(cat /tmp/b27s4/R-$TAG); . /tmp/b27s4/lib.sh; L=$R/somvr.log
    for try in 1 2 3; do
      l0=$(wc -l < $L); SYNC_S=30 sync "word practice status" "wait 1" >/dev/null
      read px pz < <(tail -n +$l0 $L | grep -aoE 'Talion known at \(-?[0-9.]+ -?[0-9.]+ -?[0-9.]+\)' | tail -1 | grep -oE -- '-?[0-9.]+' | awk 'NR==1{x=$1} NR==3{printf "%d %d\n",x,$1}')
      l0=$(wc -l < $L); SYNC_S=30 sync "word verbs orcs $px $pz 4000" "wait 1" >/dev/null
      C=$(tail -n +$l0 $L | grep -aE "^VERBS orcs t=" | grep -a "${PICK:-Beckon}" | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing.* ([0-9]+) cm from Talion.*/\1 \2 \3 \4/' | sort -k4 -n | head -1)
      echo "$(date +%T) try $try: Talion $px $pz; the nearest beast/AI (not the beckoning one): ${C:-none}"
      [ -n "$C" ] || { sleep 3; continue; }
      set -- $C; B0=$(stat -c %s $L)
      SYNC_S=40 sync "go $(( $1 + 200 )) $(( $2 + 80 )) $3" "wait 2" "word ride face on" "word hands test L -25 -35 30 0 0 0" "wait 2" >/dev/null
      for i in $(seq 1 16); do tail -c +$B0 $L | grep -aqE "the game offers the mount" && break; sleep 0.5; done
      tail -c +$B0 $L | grep -aE "^RIDE t=" | tail -1 | cut -c1-150
      SYNC_S=30 sync "word hands squeeze L 1" "wait 3" >/dev/null
      tail -c +$B0 $L | grep -aE "^RIDE t=.*(mounted|seated|dismounted)" | tail -3 | cut -c1-160
      if tail -c +$B0 $L | grep -aE "^RIDE t=.*(mounted|seated on|dismounted|off: standing)" | tail -1 | grep -q 'seated on'; then
        SYNC_S=30 sync "word ride face off" "word hands squeeze L 0" "word hands test off" "wait 1" >/dev/null; echo "$(date +%T) seated"; exit 0; fi
      SYNC_S=30 sync "word hands squeeze L 0" "wait 1" >/dev/null
    done
    echo "$(date +%T) not remounted"; exit 1
