### lib.sh
    # lib.sh - sourced by the som-b33-story6 loops (story26-steps.txt's lib.sh, /tmp/b33s6): TAG, R set by the caller
    Q=/tmp/b33s6/q.sh
    userin() { [ "$(systemctl --user is-active somvr-headset 2>/dev/null)" = active ] && return 0
      grep -q '"user": true' /run/user/$(id -u)/vr-session/status.json 2>/dev/null && return 0; return 1; }
    sync() { local c0 k; c0=$(grep -ac '^STORY status:' $R/somvr.log); $Q $TAG "$@" "word story"
      for k in $(seq 1 ${SYNC_S:-180}); do sleep 1; userin && { echo "USER IN VR: stop"; echo quit >> $R/cmd; exit 3; }
        [ "$(grep -ac '^STORY status:' $R/somvr.log)" -gt "$c0" ] && return 0; done; echo "no sync"; return 1; }
    markers() { local l0; l0=$(wc -l < $R/somvr.log); sync "word practice status nav" >/dev/null || return 1
      tail -n +$l0 $R/somvr.log | grep -aE "^PRACTICE nav [0-9]+ '$1'" | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\), [0-9.]+ m' | sed -E 's/at \(//; s/\),//; s/ m$//' | sort -k4 -n; }
    state() { grep -aE '^SESSION ui ' $R/somvr.log | tail -1 | cut -c12-80; }
    wd() { userin && { echo "USER IN VR: stop"; echo quit >> $R/cmd; exit 3; }; printf '%s' "$1" > $R/somvr.trigger; sleep ${2:-0.45}; }
    xt() { local x; { echo "# s6 $1"; shift; for x in "$@"; do echo "$x"; done; } > $R/xrtest.tmp && mv $R/xrtest.tmp $R/xrtest.txt; sleep 0.3; }
    dump() { xt dump "dump"; sleep 1.2; echo "DUMP $1 $(grep -ao 'dump layers frame [0-9]*' $R/xr-layer.log 2>/dev/null | tail -1 | grep -o '[0-9]*$')" | tee -a /tmp/b33s6/dumps-$TAG.txt; }
    win() { local a b; a=$(grep -an 'window 0 -> 1' $R/somvr.log | tail -1 | cut -d: -f1); b=$(grep -an 'window 1 -> 0' $R/somvr.log | tail -1 | cut -d: -f1); [ "${a:-0}" -gt "${b:-0}" ]; }
    tpos() { local l0; l0=$(wc -l < $R/somvr.log); wd "practice status" 1.2; tail -n +$l0 $R/somvr.log | grep -aoE 'Talion known at \(-?[0-9.]+ -?[0-9.]+ -?[0-9.]+\)' | tail -1 | grep -oE -- '-?[0-9.]+' | awk 'NR==1{x=$1} NR==2{y=$1} NR==3{printf "%d %d %d\n",x,y,$1}'; }
    news() { tail -n +$1 $R/somvr.log | grep -aE "^STORY t=.*subtitle ch=[01] '|^HINTS tutorial #.*show|^USE t=|Mission|SUCCESS|^OBJECTIVE t=.*marker [0-9]+ m=.*shown by the game|^RIDE t=.*(offers|mounted|seated on|dismounted)|^WARN|^BOW (headkill|hit check)|QuestFailed|first sight" | cut -c1-170 | awk '!s[substr($0,index($0," ")+18)]++' | head -${2:-8}; }

### gate.sh
    #!/bin/bash
    # gate.sh - som-b33-story6: wait for the boot lock (fair turns: 3 min after our last release), then the VR gate; exit 0 = ours
    ME=som-b33-story6
    while :; do
      last=$(stat -c %Y /tmp/b33s6/released 2>/dev/null || echo 0); while [ $(( $(date +%s) - last )) -lt 180 ]; do sleep 5; done
      until mkdir /tmp/somvr-boot.lock 2>/dev/null; do sleep 1; done
      echo "$ME $(date +%T)" > /tmp/somvr-boot.lock/owner
      ok=1
      [ "$(systemctl --user is-active somvr-headset 2>/dev/null)" = active ] && { echo "gate: user in VR"; ok=0; }
      grep -q '"user": false' /run/user/$(id -u)/vr-session/status.json 2>/dev/null || { echo "gate: status.json user not false"; ok=0; }
      for i in $(seq 1 240); do pgrep -x monado-service >/dev/null || break; sleep 0.5; done
      pgrep -x monado-service >/dev/null && { echo "gate: monado-service running"; ok=0; }
      pgrep -f '[H]and of Fate 2\.' >/dev/null && { echo "gate: HoF2 running"; ok=0; }
      journalctl -k -b --no-pager 2>/dev/null | grep -qiE 'amdgpu.*(reset|VRAM is lost)' && { echo "gate: amdgpu reset"; grep -q $ME /tmp/somvr-boot.lock/owner && rm -rf /tmp/somvr-boot.lock; exit 4; }
      [ $ok = 1 ] && { echo "gate: ok $(date +%T)"; exit 0; }
      grep -q $ME /tmp/somvr-boot.lock/owner && rm -rf /tmp/somvr-boot.lock; sleep 20
    done

    #!/bin/bash
    # boot.sh N - som-b33-story6: the gate + the lock, then the WINDOWS boot (headless_vr.sh --win) from a copy of $SAVE_FROM,
    # the director ($DIRECTOR, by environment) beside it (run.sh); after the boot the prefix's save copied to RUN/saves/; release

    PS="/home/bara/Projects/flat2vr/ShadowOfMordor/logs/winport/prefix/pfx/drive_c/users/steamuser/Documents/WB Games/Shadow of Mordor/Saves/GameData.sav"
    if [ "${GATE_DONE:-0}" = 1 ]; then grep -q $ME /tmp/somvr-boot.lock/owner || { echo "STOP: the lock is not ours"; exit 1; }; else /tmp/b33s6/gate.sh; rc=$?; [ $rc = 0 ] || { echo "no gate (rc $rc): stop - never boot without the lock"; exit $rc; }; fi
    cd $WT
    ( sleep 8; /tmp/b33s6/run.sh q$N "$DIRECTOR" > /tmp/b33s6/dir-q$N.log 2>&1 ) &
    HEADLESS_SAVE_FROM=${SAVE_FROM:?} HEADLESS_HOLD=${HOLD:-2400} SOMVR_EXTRA="$(grep -E '^SOMVR_[A-Z0-9_]+=' scripts/headset-build26.env | tr '\n' ' ') SOMVR_PRACTICE=1 SOMVR_PRACTICE_QUIET=0 SOMVR_PRACTICE_CLEAR=0 SOMVR_PRACTICE_DAY=0 SOMVR_HANDS_TEST=1 SOMVR_XR_TEST=@RUN@/xrtest.txt SOMVR_EYE_SIZE=2015x2217 SOMVR_BOW_HITLOG=1 ${BOOT_EXTRA:-}" setsid -w scripts/headless_vr.sh --win $WT/mod/somvr > /tmp/b33s6/boot$N.log 2>&1
    echo "boot exit $?"
    RUN=$(grep -o '^windows: [^ ]*' /tmp/b33s6/boot$N.log | head -1 | cut -c10-)

    grep -q $ME /tmp/somvr-boot.lock/owner 2>/dev/null && rm -rf /tmp/somvr-boot.lock; touch /tmp/b33s6/released; echo "released $(date +%T)"
    wait

    #!/bin/bash
    # graug2.sh TAG - som-b33-story6 boot 3 (Windows): Torvin's Graug mission start to end. The start, the follow (t2.sh), the
    # bleed (the blade's low thrust crouched behind the eating graug: SOMVR_GRAUG_BLEED), the caragor (the left hand out, the
    # grip closed AT the game's ~1.9 s offer beside a stunned beast - cont.sh's way), "Lure the Graug to Torvin" (lead it with
    # the graug 0-25 m behind, then round Torvin at ~15 m while it comes, never standing), Step Three (the VR bow at the graug's
    # head from the saddle, `bow aimhead` = the test's aim), thrown off = mount again; the end's windows; next.sh.
    TAG=$1; R=$(cat /tmp/b33s6/R-$TAG); . /tmp/b33s6/lib.sh; L=$R/somvr.log; echo $$ > /tmp/b33s6/pid-g4-$TAG
    cnt() { grep -acE "$1" "$L"; }
    d2() { awk -v a=$1 -v b=$2 -v x=$3 -v z=$4 'BEGIN{print int(sqrt((a-x)^2+(b-z)^2))}'; }
    PRE_L="hands test R 20 -45 -18 -30 0 0"; THRUST_L="hands swing R 10 -85 50 -30 0 0 180"
    draw_sword() { wd "hands test R 10 -10 -15 0 0 0" 0.7; wd "hands squeeze R 1" 0.6; wd "hands test R 30 -25 40 0 -90 0" 0.6; wd "hands squeeze R 0" 0.5; }
    touch /tmp/b33s6/hold-$TAG; wd "story invisible off" 0.4
    if [ "${SKIP_START:-0}" = 0 ]; then echo "PHASE start $(date +%T)"
      SYNC_S=200 sync "go 23266 3500 -19020" "wait 3" "pics s6-marker" "tap R b 0.4" "wait 3" "pics s6-card" "tap R a 0.4" "wait 4" >/dev/null
      echo "PHASE follow $(date +%T)"; STOP_RE="show 'GP_Torvin02_Tut_TrollBleed'" /tmp/b33s6/t2.sh $TAG ${MIN1:-12}
    fi
    # ---------------- the bleed ----------------
    wd "story invisible on" 0.5; draw_sword
    graug() { local l0 TM mx my mz; l0=$(wc -l < $L); wd "objective status" 1.2
      TM=$(tail -n +$l0 $L | grep -aE "^OBJECTIVE status marker .*'Troll'" | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
      read mx my mz <<< "${TM:-32281 2209 -27717}"; l0=$(wc -l < $L); wd "verbs orcs $mx $mz 3500" 1.6; wd "verbs orcs 32281 -27717 3500" 1.6
      tail -n +$l0 $L | grep -aE "^VERBS orcs t=.*attack '(Trl|AI_Trl|AI_Troll|Troll)" | sort -t"'" -k2 | grep -a Eating | head -1 | sed -E 's/.* orc (0x[0-9a-f]+) at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing (-?[0-9.]+) deg.* attack .(.*)./\1 \2 \3 \4 \5 \6/'; }
    round() { local nm=$1 D=$2 k G gc gx gy gz gf ga tx tz B0
      for k in $(seq 1 40); do G=$(graug); echo "$G" | grep -q 'Eating' && break; sleep 4; done
      [ -n "$G" ] || { echo "$(date +%T) $nm: no graug"; return 1; }; read gc gx gy gz gf ga <<< "$G"
      read tx tz < <(python3 -c "import math;f=math.radians($gf);print(round($gx-$D*math.sin(f)),round($gz-$D*math.cos(f)))")
      B0=$(wc -l < $L); wd "practice go $tx $((gy - 150)) $tz" 3.0; wd "ride face graug" 2.5; wd "ride face off" 0.3
      wd "crouch hold 9000" 1.0; draw_sword; wd "$PRE_L" 0.8; dump s6-$nm-ready; sleep 0.5
      wd "$THRUST_L" 1.0; dump s6-$nm-in; sleep 2.5; wd "$PRE_L" 0.6; sleep 2; dump s6-$nm-after
      echo "$(date +%T) $nm: graug $gc ($ga) at $gx $gz facing $gf, Talion $D cm behind: $(tail -n +$B0 $L | grep -aE 'VERBS t=.*bleed' | head -1 | cut -c1-200)"
      tail -n +$B0 $L | grep -aqE "Talion attack 'PC_Hamstring"; }
    [ "${SKIP_BLEED:-0}" = 1 ] || { echo "PHASE bleed $(date +%T)"; HS=1
    for D in 140 130 150 135 125; do round B$D $D && { HS=0; break; }; sleep 8; done; echo "bleed: hamstring $([ $HS = 0 ] && echo yes || echo no) $(date +%T)"
    }; wd "crouch stand" 0.3; wd "story invisible off" 0.4
    # ---------------- the mount: the hand out, the grip closed at the offer ----------------
    mount() { local T B0 B1 K kd C P cx cy cz lastgo=0 got=0
      wd "ride steer off" 0.2; xt off; draw_sword
      wd "hands test R 30 -10 45 0 -35 0" 0.4; wd "fight guard learn" 0.4; wd "ride guard on" 0.3; wd "ride face on" 0.3
      wd "hands test L -25 -35 30 0 0 0" 0.5; wd "hands squeeze L 0" 0.5
      T=$((SECONDS + ${1:-8} * 60)); B0=$(stat -c %s $L)
      while [ $SECONDS -lt $T ]; do
        if tail -c +$B0 $L | grep -aq 'the game offers the mount'; then
          printf '%s' "hands squeeze L 1" > $R/somvr.trigger; B1=$(stat -c %s $L); echo "$(date +%T) the offer: the grip closed"

          sleep 1; wd "hands squeeze L 0" 0.4; B0=$(stat -c %s $L); continue; fi
        win && { sleep 1.5; dump m-w; xt a1 "btn R a 1"; sleep 0.3; xt a0 "btn R a 0"; sleep 2; xt off; B0=$(stat -c %s $L); }
        if state | grep -q 'ctx=struggle'; then echo "$(date +%T) a struggle: hard thrusts"

          wd "hands test R 30 -10 45 0 -35 0" 0.3; fi
        K=$(tail -c 60000 $L | grep -aE "^BEASTS t=.*caragor '[A-Za-z_]+' Combat=0x[0-9a-f]+ attack 'AI_Warg_Knockdown_Idle" | tail -1)
        if [ -n "$K" ] && [ $((SECONDS - lastgo)) -gt 6 ]; then kd=$(echo "$K" | grep -oE 'at [0-9.]+ m' | grep -oE '[0-9.]+'); C=$(echo "$K" | grep -oE 'Combat=0x[0-9a-f]+' | cut -c8-)
          if awk -v d=${kd:-9} 'BEGIN{exit !(d > 2.2)}'; then read px py pz <<< "$(tpos)"; l1=$(wc -l < $L); wd "verbs orcs $px $pz 4000" 1.0
            P=$(tail -n +$l1 $L | grep -a "orc $C at" | tail -1 | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing.*/\1 \2 \3/')

          fi; fi
        sleep 0.1
      done
      wd "ride face off" 0.2; wd "ride guard off" 0.2
      for i in $(seq 1 40); do tail -c +${B1:-1} $L | grep -aq "^RIDE t=.* seated on" && break; sleep 0.25; done; sleep 1.5

    echo "PHASE mount $(date +%T)"; mount 9; dump s6-seated
    # ---------------- the lure, round Torvin, the bow ----------------
    BOW_SLOT="hand L head -12 -10 -15 90 0 0"; BOW_STRAIGHT="hand L head -8 -15 55 90 0 0"
    AT_STRING="hand R head -5 -13 30 90 0 0"; DRAWN="hand R head 3 -10 -3 90 0 0"; R_REST="hand R head 22 -40 15 0 0 0"
    xl() { userin && { echo "USER IN VR"; echo quit >> $R/cmd; exit 3; }; { echo "# $(date +%s%N)"; echo "head 0 0 0 yaw ${HY:-0} pitch ${HP:-0}"; for l in "$@"; do echo "$l"; done; } > "$R/xrtest.tmp" && mv "$R/xrtest.tmp" "$R/xrtest.txt"; sleep "${TS:-0.8}"; }

      xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1"; sleep 0.4; xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1" "btn R trigger 1"; sleep 0.5
      if [ "${QUICK:-0}" = 1 ]; then TS=0.3 xl "$BOW_STRAIGHT" "$DRAWN s 0.2" "btn L squeeze 1" "btn R trigger 1"; else xl "$BOW_STRAIGHT" "$DRAWN s 0.8" "btn L squeeze 1" "btn R trigger 1"; sleep 1.3; fi; }

    GC=""; TV="22100 2350 -26650"; LB=$(wc -l < $L); MODE=lead; ANG=0; SHOTS=0; STEP3=0; AB=0; STUNS0=$(cnt "^BOW graugstun t="); T=$((SECONDS + ${MIN_LURE:-16} * 60)); n=0; THROWN=0

      l1=$(wc -l < $L); wd "verbs orcs ${px:-0} ${pz:-0} 5000" 1.0

    #!/bin/bash
    # probe.sh TAG DIST ANGLE N - som-b33-story6: Talion DIST cm from the graug on the bearing ANGLE (deg, 0 = away from Torvin),
    # then N bow shots at its head (`bow aimhead` = the test's aim), each shot's hit attack printed
    TAG=$1; D=${2:-2000}; A=${3:-0}; N=${4:-4}; R=$(cat /tmp/b33s6/R-$TAG); . /tmp/b33s6/lib.sh; L=$R/somvr.log
    BOW_SLOT="hand L head -12 -10 -15 90 0 0"; BOW_STRAIGHT="hand L head -8 -15 55 90 0 0"
    AT_STRING="hand R head -5 -13 30 90 0 0"; DRAWN="hand R head 3 -10 -3 90 0 0"; R_REST="hand R head 22 -40 15 0 0 0"
    xl() { { echo "# $(date +%s%N)"; echo "head 0 0 0 yaw 0 pitch 0"; for l in "$@"; do echo "$l"; done; } > "$R/xrtest.tmp" && mv "$R/xrtest.tmp" "$R/xrtest.txt"; sleep "${TS:-0.8}"; }
    bow_ok() { grep -aE '^BOW t=.* (enter |exit: )' "$L" | tail -1 | grep -q ' enter '; }
    bow_out() { wd "hands squeeze L -1" 0.3; wd "hands squeeze R -1" 0.3; wd "hands test off" 0.4; xl "$BOW_SLOT" "$R_REST"; sleep 0.3; xl "$BOW_SLOT" "$R_REST" "btn L squeeze 1"; sleep 0.8; xl "$BOW_STRAIGHT" "$R_REST" "btn L squeeze 1"; sleep 1; }
    bow_draw() { bow_ok || { xl "$BOW_SLOT" "$R_REST" "btn L squeeze 0"; sleep 0.6; bow_out; }
      xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1"; sleep 0.4; xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1" "btn R trigger 1"; sleep 0.5
      xl "$BOW_STRAIGHT" "$DRAWN s 0.8" "btn L squeeze 1" "btn R trigger 1"; sleep 1.3; }
    bow_loose() { TS=0.1 xl "$BOW_STRAIGHT" "$DRAWN" "btn L squeeze 1" "btn R trigger 0"; }
    GC=$(grep -aE "^BEASTS t=.*graug '[A-Za-z_]+' Combat=0x[0-9a-f]+ .*\(health [0-9]+\)" $L | awk -F'health ' '{h=$2+0; if (h < 1700) print}' | tail -1 | grep -oE 'Combat=0x[0-9a-f]+' | cut -c8-)
    gpos() { local l1; read px py pz <<< "$(tpos)"; l1=$(wc -l < $L); wd "verbs orcs ${px:-0} ${pz:-0} 5000" 1.0
      tail -n +$l1 $L | grep -a "orc $GC at" | head -1 | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing (-?[0-9.]+) deg.* ([0-9]+) cm from Talion.*attack .(.*)./\1 \2 \3 \4 \5 \6/'; }
    read gx gy gz gf gd ga <<< "$(gpos)"; echo "$(date +%T) the graug $GC at $gx $gz facing $gf, $gd cm, $ga"
    read tx tz < <(python3 -c "
    import math; ox,oz=22100,-26650; vx,vz=$gx-ox,$gz-oz; b=math.atan2(vx,vz)+math.radians($A); print(round($gx+$D*math.sin(b)), round($gz+$D*math.cos(b)))")
    wd "practice go $tx $((gy + 100)) $tz" 3.0; echo "$(date +%T) Talion to $tx $tz: $(grep -a 'arrived at\|did not arrive' $L | tail -1 | cut -c1-100)"
    wd "bow arrows 8" 0.4
    for k in $(seq 1 $N); do read gx gy gz gf gd ga <<< "$(gpos)"; bow_draw; n1=$(wc -l < $L); wd "bow aimhead $gx $gy $gz" 0.2; bow_loose; sleep 2.5
      echo "$(date +%T) shot $k: the graug $gd cm ($ga, facing $gf): $(tail -n +$n1 $L | grep -aE '^BOW hit t=' | grep -oE "the attack '[^']+'|record '[^']+'" | tr '\n' ' ') $(tail -n +$n1 $L | grep -aE '^BOW hit check' | grep -oE "health [0-9.]+ -> [0-9.]+" | head -1)"
    done
    news $(( $(wc -l < $L) - 3000 )) 8 | grep -avE "Over here|Look here|Hey|beard|morsel|so close|filthy|Spiders|ugly|Come and get"

    bow_draw() { bow_ok || { xl "$BOW_SLOT" "$R_REST" "btn L squeeze 0"; sleep 0.6; bow_out; }
      xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1"; sleep 0.4; xl "$BOW_STRAIGHT" "$AT_STRING" "btn L squeeze 1" "btn R trigger 1"; sleep 0.5
      xl "$BOW_STRAIGHT" "$DRAWN s 0.8" "btn L squeeze 1" "btn R trigger 1"; sleep 1.3; }
    bow_loose() { TS=0.1 xl "$BOW_STRAIGHT" "$DRAWN" "btn L squeeze 1" "btn R trigger 0"; }
    GC=$(grep -aE "^BEASTS t=.*graug '[A-Za-z_]+' Combat=0x[0-9a-f]+ .*\(health [0-9]+\)" $L | awk -F'health ' '{h=$2+0; if (h < 1700) print}' | tail -1 | grep -oE 'Combat=0x[0-9a-f]+' | cut -c8-)
    gpos() { local l1; read px py pz <<< "$(tpos)"; l1=$(wc -l < $L); wd "verbs orcs ${px:-0} ${pz:-0} 5000" 1.0
      tail -n +$l1 $L | grep -a "orc $GC at" | head -1 | sed -E 's/.* at (-?[0-9]+),(-?[0-9]+),(-?[0-9]+) facing (-?[0-9.]+) deg.* ([0-9]+) cm from Talion.*attack .(.*)./\1 \2 \3 \4 \5 \6/'; }
    read gx gy gz gf gd ga <<< "$(gpos)"; echo "$(date +%T) the graug $GC at $gx $gz facing $gf, $gd cm, $ga"
    read tx tz < <(python3 -c "
    import math; ox,oz=22100,-26650; vx,vz=$gx-ox,$gz-oz; b=math.atan2(vx,vz)+math.radians($A); print(round($gx+$D*math.sin(b)), round($gz+$D*math.cos(b)))")
    [ "${NOGO:-0}" = 1 ] || wd "practice go $tx $((gy + 100)) $tz" 3.0; echo "$(date +%T) Talion to $tx $tz: $(grep -a 'arrived at\|did not arrive' $L | tail -1 | cut -c1-100)"
    wd "bow arrows 8" 0.4
    for k in $(seq 1 $N); do read gx gy gz gf gd ga <<< "$(gpos)"; bow_draw; n1=$(wc -l < $L); wd "bow aimhead $gx $gy $gz" 0.2; bow_loose; sleep 2.5
      sleep 1.5; echo "$(date +%T) shot $k: the graug $gd cm ($ga): $(tail -n +$n1 $L | grep -aE '^BOW (hit t=|graugstun t=)' | grep -oE "the attack '[^']+'|record '[^']+'|graugstun t=[0-9.]+: [a-z -]+|held full [0-9.]+ s" | tr '\n' ' ')"
      tail -n +$n1 $L | grep -aE "^BEASTS t=.*graug .*Combat=$GC|^VERBS t=.*orc $GC" | grep -oE "^[A-Z]+ t=[0-9.]+|attack '[^']*'|state '[^']*'|health [0-9.]+" | paste -sd' ' | cut -c1-400
    done
    news $(( $(wc -l < $L) - 3000 )) 8 | grep -avE "Over here|Look here|Hey|beard|morsel|so close|filthy|Spiders|ugly|Come and get"
