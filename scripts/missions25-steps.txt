## the loops (bash; R = the run dir; q.sh appends commands to the driver's ctl queue)
### q.sh
    #!/bin/bash
    # q.sh TAG CMD... : append commands to the driver's ctl queue (next index kept in DIR/next)
    TAG=$1; shift; D=/tmp/b25m/ctl-$TAG; n=$(cat $D/next 2>/dev/null || ls $D | grep -c '^cmd[0-9]*\.txt$'); 
    for c in "$@"; do n=$((n+1)); printf '%s\n' "$c" > $D/.t && mv $D/.t $D/cmd$n.txt; done; echo $n > $D/next
### follow.sh
    #!/bin/bash
    # follow.sh TAG ROUNDS NAME_RE [DY] - teleport after the objective marker NAME_RE (status, then go there), ROUNDS times
    TAG=$1; N=$2; RE=$3; DY=${4:-100}; R=$(cat /tmp/b25m/R-$TAG)
    sync() { local c0; c0=$(grep -c '^STORY status:' $R/somvr.log); /tmp/b25m/q.sh $TAG "$@" "word story"
      for k in $(seq 1 90); do sleep 1; [ "$(grep -c '^STORY status:' $R/somvr.log)" -gt "$c0" ] && return 0; done; return 1; }
    for i in $(seq 1 $N); do
      l0=$(wc -l < $R/somvr.log); sync "status" || { echo "no sync"; break; }
      p=$(tail -n +$l0 $R/somvr.log | grep -E "^OBJECTIVE status marker .*'$RE'" | tail -1 | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\)' | tr -d '()' | cut -c4-)
      nm=$(tail -n +$l0 $R/somvr.log | grep -E "^OBJECTIVE status marker" | grep -oE "'[A-Za-z_0-9]+'[^:]*at \([-0-9 ]*\)" | tr '\n' ';')
      sub=$(tail -n +$l0 $R/somvr.log | grep -E '^STORY t=.*subtitle ch=1 .[A-Z]' | tail -1 | cut -d"'" -f2)
      echo "$(date +%T) round $i: $nm | $sub"
      [ -z "$p" ] && { echo "no $RE marker: stop"; break; }
      set -- $p; sync "go $1 $(( $2 + DY )) $3" "wait ${WAIT:-3}" || break
    done
### archers.sh
    #!/bin/bash
    # archers.sh TAG [MAX] - the Sea Fort's archers: their 'ObjectiveGold' places from `practice status nav`, then for each:
    # a teleport onto it and the driver's `brand 100` (crouched, behind the nearest orc, the palm at its head, the left trigger)
    TAG=$1; MAX=${2:-8}; R=$(cat /tmp/b25m/R-$TAG)
    sync() { local c0; c0=$(grep -c '^STORY status:' $R/somvr.log); /tmp/b25m/q.sh $TAG "$@" "word story"
      for k in $(seq 1 120); do sleep 1; [ "$(grep -c '^STORY status:' $R/somvr.log)" -gt "$c0" ] && return 0; done; return 1; }
    for round in $(seq 1 $MAX); do
      l0=$(wc -l < $R/somvr.log); sync "word practice status nav"
      mapfile -t P < <(tail -n +$l0 $R/somvr.log | grep -E "^PRACTICE nav [0-9]+ 'ObjectiveGold'" | grep -oE 'at \(-?[0-9]+ -?[0-9]+ -?[0-9]+\), [0-9.]+ m' | sed -E 's/at \(//; s/\),//; s/ m$//' | sort -k4 -n)
      echo "$(date +%T) round $round: ${#P[@]} archer markers; branded so far: $(grep -cE "PC_(Archer|Bip)_StealthBrand" $R/somvr.log) brand lines; $(grep -E "subtitle ch=1 '[0-9] / 6|Brand Archers" $R/somvr.log | tail -1 | cut -c1-80)"
      [ ${#P[@]} -eq 0 ] && { echo "no archer markers: stop"; break; }
      set -- ${P[0]}; echo "  the nearest: $1 $2 $3 ($4 m)"
      l1=$(wc -l < $R/somvr.log); sync "gobrand $1 $(( $2 + 50 )) $3" "wait 5"
      tail -n +$l1 $R/somvr.log | grep -E "VERBS brand t=.*(press|started|offer)|attack 'PC_(Archer|Bip)_StealthBrand|Mission Failed|subtitle ch=1" | head -4 | cut -c1-200
    done

    sync "word hands test R 10 -10 -15 0 0 0" "word hands squeeze R 1" "word hands test R 30 -25 40 0 -90 0" "word hands squeeze R 0"
    for round in $(seq 1 $MAX); do
